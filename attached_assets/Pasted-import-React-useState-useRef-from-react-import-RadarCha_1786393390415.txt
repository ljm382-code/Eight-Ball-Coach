function newSkillState() {
  return { rating: 30, attempts: [] };
}
function newProfile() {
  const skills = {};
  SKILLS.forEach((s) => (skills[s.id] = newSkillState()));
  return { dataVersion: CONFIG.dataVersion, assessmentComplete: false, skills, ratingHistory: {}, rootCauseTally: {}, sessions: [] };
}
function migrateProfile(raw) {
  if (!raw.dataVersion) raw.dataVersion = CONFIG.dataVersion;
  if (!raw.ratingHistory) raw.ratingHistory = {};
  if (!raw.rootCauseTally) raw.rootCauseTally = {};
  return raw;
}

function computeConfidence(attempts) {
  if (attempts.length === 0) return { score: 0, tier: "Low" };
  const cfg = CONFIG.confidence;
  const weightedN = attempts.reduce((sum, a) => sum + (a.source === "assessment" ? cfg.assessmentWeight : 1), 0);
  const distinctDrills = new Set(attempts.map((a) => a.drillId)).size;
  const now = Date.now();
  const recentCount = attempts.filter((a) => now - a.ts < CONFIG.recencyWindowMs).length;
  const volumeScore = Math.min(1, weightedN / cfg.volumeCap);
  const diversityScore = Math.min(1, distinctDrills / cfg.diversityCap);
  const recencyScore = Math.min(1, recentCount / cfg.recentCap);
  const gatedVolume = volumeScore * (cfg.diversityFloor + (1 - cfg.diversityFloor) * diversityScore);
  const score = clamp(gatedVolume * cfg.weights.volumeDiversity + recencyScore * cfg.weights.recency, 0, 1);
  let tier = "Low";
  if (score >= cfg.tiers.high) tier = "High";
  else if (score >= cfg.tiers.established) tier = "Established";
  else if (score >= cfg.tiers.emerging) tier = "Emerging";
  return { score, tier };
}
function isStale(skillState) {
  if (skillState.attempts.length === 0) return false;
  const last = skillState.attempts[skillState.attempts.length - 1];
  return Date.now() - last.ts > CONFIG.recencyWindowMs * 2;
}
function displayTier(tier, stale) {
  if (!stale) return tier;
  const downgrade = { High: "Established", Established: "Emerging", Emerging: "Low", Low: "Low" };
  return downgrade[tier];
}
function confidenceLabel(tier, stale) {
  const t = displayTier(tier, stale);
  if (t === "Low") return stale ? "Uncertain — hasn't been tested recently" : "Still learning your game";
  if (t === "Emerging") return "Getting a clearer picture";
  return "Strong evidence";
}

// Raw execution evidence ALWAYS updates the observed (task) skill — hardening #1.
// Root-cause inference never redirects this; it only feeds diagnosis (see evidenceForSkill/rootCauseTally).
function applySkillUpdate(profile, skillId, value, meta = {}) {
  const s = profile.skills[skillId];
  const now = Date.now();
  const difficulty = meta.difficulty ?? 5;
  const conf = computeConfidence(s.attempts);
  const kBase = CONFIG.kMax - (CONFIG.kMax - CONFIG.kMin) * conf.score;
  const diffOffset = (difficulty - CONFIG.difficultyMidpoint) * CONFIG.difficultySensitivity;
  let diffMultiplier = value >= 0.7 ? 1 + diffOffset : 1 - diffOffset;
  diffMultiplier = clamp(diffMultiplier, CONFIG.diffMultiplierMin, CONFIG.diffMultiplierMax);
  const effectiveK = clamp(kBase * diffMultiplier, CONFIG.kMin * 0.5, CONFIG.kMax * 1.5);
  const rating = clamp(s.rating + effectiveK * (value * 100 - s.rating), 0, 100);
  const attempt = { ts: now, value, difficulty, ...meta };
  return { ...profile, skills: { ...profile.skills, [skillId]: { rating, attempts: [...s.attempts, attempt] } } };
}
function bumpRootCauseTally(profile, skillId) {
  const tally = { ...(profile.rootCauseTally || {}) };
  tally[skillId] = (tally[skillId] || 0) + 1;
  return { ...profile, rootCauseTally: tally };
}

function resultValue(execResult) {
  if (execResult === "success") return 1;
  if (execResult === "partial") return 0.5;
  return 0;
}
function decisionValue(tier) {
  if (tier === "optimal") return 1;
  if (tier === "acceptable") return 0.7;
  if (tier === "highrisk") return 0.4;
  return 0;
}

function composite(profile, type) {
  const ids = SKILLS.filter((s) => s.type === type).map((s) => s.id);
  const vals = ids.map((id) => profile.skills[id].rating);
  return vals.reduce((a, b) => a + b, 0) / vals.length;
}
function overallMean(profile) {
  const vals = SKILLS.map((s) => profile.skills[s.id].rating);
  return vals.reduce((a, b) => a + b, 0) / vals.length;
}

function evidenceForSkill(profile, skillId) {
  const st = profile.skills[skillId];
  const mean = overallMean(profile);
  const gap = mean - st.rating;
  const conf = computeConfidence(st.attempts);
  const now = Date.now();
  const recent = st.attempts.filter((a) => now - a.ts < CONFIG.recencyWindowMs);
  const recentFails = recent.filter((a) => a.value < 0.5).length;
  const recentErrorRate = recent.length ? recentFails / recent.length : 0;
  const clearanceFails = st.attempts.filter((a) => a.clearance && a.value < 0.5).length;
  const rootCauseCount = (profile.rootCauseTally || {})[skillId] || 0;
  const rootCauseScore = Math.min(1, rootCauseCount / 3);
  const w = CONFIG.evidence.weights;
  const score =
    clamp(gap / 25, 0, 1) * w.gap +
    conf.score * w.confidence +
    recentErrorRate * w.recentError +
    Math.min(1, clearanceFails / 3) * w.clearanceFail +
    rootCauseScore * w.rootCause;
  let status = "none";
  // A skill can qualify either through its own weak rating OR through repeated root-cause evidence
  // even if its own execution attempts look fine (e.g. rarely attempted directly but frequently implicated).
  const qualifies = conf.tier !== "Low" && (gap > CONFIG.meanGapThreshold || rootCauseCount >= 2);
  if (qualifies) {
    if (score >= CONFIG.evidence.confirmedThreshold) status = "confirmed";
    else if (score >= CONFIG.evidence.provisionalThreshold) status = "provisional";
  }
  return { rating: st.rating, gap, confidence: conf, score, status, rootCauseCount };
}
function limitingFactor(profile) {
  const scored = SKILLS.map((s) => ({ ...s, ...evidenceForSkill(profile, s.id) })).filter((s) => s.status !== "none");
  if (scored.length === 0) return { primary: null, secondary: null, status: "insufficient" };
  scored.sort((a, b) => b.score - a.score);
  return { primary: scored[0], secondary: scored[1] || null, status: scored[0].status };
}
function sessionWeighting(profile) {
  const exec = composite(profile, "execution");
  const dec = composite(profile, "decision");
  const gap = dec - exec;
  const execWeight = clamp(50 + gap * 0.6, 25, 75);
  return { execWeight: Math.round(execWeight), decWeight: Math.round(100 - execWeight), exec, dec };
}
function difficultyForSkill(profile, skillId) {
  const rating = profile.skills[skillId].rating;
  return clamp(Math.round(rating / 12), 1, 8);
}
function nearestDrill(pool, targetDiff) {
  return pool.slice().sort((a, b) => Math.abs(a.difficulty - targetDiff) - Math.abs(b.difficulty - targetDiff))[0];
}

/* ---------------------------------------------------------------
   ERROR-EVIDENCE LAYER — raw evidence preserved, diagnosis layered
   on top (hardening #1, #2, #3)
--------------------------------------------------------------- */
// shotEntries: prior shots in this clearance, each { observedSkill, value, reportedError, ballId }
function inferLikelyCause(shotEntriesBefore, currentObservedSkill, currentReportedError) {
  const prev1 = shotEntriesBefore[shotEntriesBefore.length - 1];
  const prev2 = shotEntriesBefore[shotEntriesBefore.length - 2];

  if (currentReportedError === "MISS" && prev1?.reportedError === "POSITION" && prev2?.reportedError === "SPEED") {
    return { skill: "speed", pattern: "B", confidence: "Emerging", note: "Traced back to a speed-control error that produced the poor position." };
  }
  if (currentReportedError === "MISS" && prev1?.reportedError === "POSITION") {
    return { skill: "positional", pattern: "A", confidence: "Emerging", note: "Traced to a poor position left by the previous shot." };
  }
  if (currentReportedError === "MISS" && (prev1?.reportedError === "PATTERN" || prev1?.reportedError === "DECISION")) {
    return { skill: "pattern", pattern: "C", confidence: "Emerging", note: "Traced to an earlier pattern/decision choice that forced a difficult recovery." };
  }
  return { skill: currentObservedSkill, pattern: "D", confidence: "Low", note: "No clear upstream cause found — treated as a direct execution issue." };
}

// Distinguishes immediate / contributing / root without ever touching the raw skillId of any attempt.
function classifyErrorChain(shotEntries) {
  const failed = shotEntries.filter((e) => e.value === 0);
  if (failed.length === 0) return null;
  const last = failed[failed.length - 1];
  const lastIdx = shotEntries.indexOf(last);
  const cause = inferLikelyCause(shotEntries.slice(0, lastIdx), last.observedSkill, last.reportedError);
  const immediateSkill = last.observedSkill;
  const rootSkill = cause.skill;
  const contributingSkill = rootSkill !== immediateSkill && failed.length > 1 ? failed[failed.length - 2].observedSkill : null;
  return { immediateSkill, contributingSkill, rootSkill, rootCauseConfidence: cause.confidence, pattern: cause.pattern };
}

function buildErrorChainNarrative(chain) {
  if (!chain || chain.rootSkill === chain.immediateSkill) return null;
  return `Although the clearance ended on a ${SKILL_MAP[chain.immediateSkill]?.name.toLowerCase()} failure, the sequence first broke down through ${SKILL_MAP[chain.rootSkill]?.name.toLowerCase()}.`;
}

// hardening #6/#9: only real decision evidence (live plan/adaptation events, or an explicit
// PATTERN/DECISION report) may move a decision skill — a bare execution failure never does.
function computeClearanceDecisionScore(entries) {
  const hasLiveDecisionEvidence = entries.some((e) => e.type === "planDecision" || e.type === "adaptation");
  if (hasLiveDecisionEvidence) return null;
  const flaggedPoor = entries.some((e) => e.reportedError === "PATTERN" || e.reportedError === "DECISION");
  if (flaggedPoor) return { tier: "poor", value: decisionValue("poor"), skillId: "pattern" };
  return null;
}

// hardening #5: compare a route against preferred/acceptable routes for a clearance
function routeTier(route, clearance) {
  const key = route.join(",");
  if (key === clearance.preferredRoute.join(",")) return "optimal";
  if ((clearance.acceptableRoutes || []).some((r) => r.join(",") === key)) return "acceptable";
  const matches = route.filter((id, i) => id === clearance.preferredRoute[i]).length;
  const frac = clearance.preferredRoute.length ? matches / clearance.preferredRoute.length : 0;
  return frac >= 0.5 ? "highrisk" : "poor";
}

function resolveFailureMode(clearance, errorCode) {
  if (errorCode === "POSITION" && clearance.adaptationEligible) return "continue_from_position";
  return clearance.failureMode;
}

/* ---------------------------------------------------------------
   SESSION GENERATION (hardening #8, #9, #10, #11)
--------------------------------------------------------------- */
const SESSION_LENGTHS = { 15: 3, 30: 5, 45: 7, 60: 9, 90: 13 };

function reasonForDrill(drill, ctx) {
  if (drill.skillId === ctx.calibrationId) return `Calibration — we still need more evidence about ${SKILL_MAP[drill.skillId].name.toLowerCase()}.`;
  if (drill.skillId === ctx.maintenanceId) return `Maintenance — your ${SKILL_MAP[drill.skillId].name.toLowerCase()} is strong but hasn't been tested recently.`;
  if (ctx.focusSkillIds.includes(drill.skillId)) return `${SKILL_MAP[drill.skillId].name} is currently your main training focus.`;
  if (drill.skillId === ctx.supportingSlot) return `${SKILL_MAP[drill.skillId].name} supports the skill you're focusing on today.`;
  return "Rounding out today's session.";
}

// hardening #9: repair pass run after all substitutions
function validateSessionComposition(selected, profile, lf) {
  let result = [...selected];
  const seen = new Set();
  result = result.map((d) => {
    if (d.type === "combined") return d;
    if (seen.has(d.id)) {
      const altPool = DRILLS.filter((x) => x.type === d.type && x.skillId === d.skillId && !seen.has(x.id));
      const replacement = altPool.length ? nearestDrill(altPool, d.difficulty) : d;
      seen.add(replacement.id);
      return replacement;
    }
    seen.add(d.id);
    return d;
  });

  const hasExec = result.some((d) => d.type === "execution");
  const hasDec = result.some((d) => d.type === "decision");
  if (!hasExec && result.length > 1) {
    const pool = DRILLS.filter((d) => d.type === "execution");
    result[0] = nearestDrill(pool, difficultyForSkill(profile, pool[0].skillId));
  }
  if (!hasDec && result.length > 1) {
    const pool = DRILLS.filter((d) => d.type === "decision");
    result[result.length - 1] = nearestDrill(pool, difficultyForSkill(profile, pool[0].skillId));
  }
  if (lf.primary) {
    const hasFocus = result.some((d) => d.skillId === lf.primary.id);
    if (!hasFocus) {
      const pool = DRILLS.filter((d) => d.skillId === lf.primary.id);
      if (pool.length) result[0] = nearestDrill(pool, difficultyForSkill(profile, lf.primary.id));
    }
  }
  return result;
}

function generateSession(profile, minutes) {
  const count = SESSION_LENGTHS[minutes] || 5;
  const lf = limitingFactor(profile);
  const weighting = sessionWeighting(profile);
  let execCount = clamp(Math.round((count * weighting.execWeight) / 100), 1, count - 1);
  let decCount = count - execCount;

  const focusSkillIds = [lf.primary?.id, lf.secondary?.id].filter(Boolean);
  const supportingCandidates = focusSkillIds.flatMap((id) => SUPPORTING_LINKS.filter((l) => l.target === id).map((l) => l.source));
  const supportingSlot = supportingCandidates[0] || null;

  const execFocus = focusSkillIds.filter((id) => SKILL_MAP[id].type === "execution");
  const decFocus = focusSkillIds.filter((id) => SKILL_MAP[id].type === "decision");
  const execPool = execFocus.length ? execFocus : SKILLS.filter((s) => s.type === "execution").map((s) => s.id);
  const decPool = decFocus.length ? decFocus : SKILLS.filter((s) => s.type === "decision").map((s) => s.id);

  const usedIds = new Set();
  function pick(type, n, skillIds) {
    const chosen = [];
    for (let i = 0; i < n; i++) {
      const skillId = skillIds[i % skillIds.length];
      const targetDiff = difficultyForSkill(profile, skillId);
      const preferred = DRILLS.filter((d) => d.type === type && d.skillId === skillId && !usedIds.has(d.id));
      const anyUnused = DRILLS.filter((d) => d.type === type && !usedIds.has(d.id));
      const source = preferred.length ? preferred : anyUnused.length ? anyUnused : DRILLS.filter((d) => d.type === type);
      const drill = nearestDrill(source, targetDiff);
      usedIds.add(drill.id);
      chosen.push(drill);
    }
    return chosen;
  }

  let execSelected = pick("execution", execCount, execPool);
  let decSelected = pick("decision", decCount, decPool);

  // hardening #8: supporting skill may be execution OR decision — no longer execution-only
  if (supportingSlot) {
    const supType = SKILL_MAP[supportingSlot].type;
    const targetArr = supType === "execution" ? execSelected : decSelected;
    if (targetArr.length) {
      const pool = DRILLS.filter((d) => d.type === supType && d.skillId === supportingSlot && !usedIds.has(d.id));
      if (pool.length) {
        const idx = targetArr.length - 1;
        targetArr[idx] = nearestDrill(pool, difficultyForSkill(profile, supportingSlot));
        usedIds.add(targetArr[idx].id);
      }
    }
  }

  let selected = [...execSelected, ...decSelected];

  let maintenanceId = null;
  if (count >= 4) {
    const strongest = SKILLS.map((s) => ({ ...s, st: profile.skills[s.id] }))
      .filter((s) => computeConfidence(s.st.attempts).tier !== "Low")
      .sort((a, b) => b.st.rating - a.st.rating)[0];
    if (strongest) {
      const pool = DRILLS.filter((d) => d.skillId === strongest.id);
      if (pool.length && selected.length) {
        selected[0] = nearestDrill(pool, difficultyForSkill(profile, strongest.id));
        maintenanceId = strongest.id;
      }
    }
  }

  let calibrationId = null;
  if (count >= 6) {
    const leastObserved = SKILLS.map((s) => ({ ...s, n: profile.skills[s.id].attempts.length, tier: computeConfidence(profile.skills[s.id].attempts).tier }))
      .filter((s) => s.tier === "Low" && !focusSkillIds.includes(s.id))
      .sort((a, b) => a.n - b.n)[0];
    if (leastObserved) {
      const pool = DRILLS.filter((d) => d.skillId === leastObserved.id);
      const idx = selected.length > 1 ? 1 : 0;
      if (pool.length) {
        selected[idx] = nearestDrill(pool, difficultyForSkill(profile, leastObserved.id));
        calibrationId = leastObserved.id;
      }
    }
  }

  if (count >= 5) {
    const clearance = weighting.execWeight >= 50 ? CLEARANCES.find((c) => c.id === "clr4") : CLEARANCES.find((c) => c.id === "clr5");
    selected = [...selected, { ...clearance, reason: "Clearance work — combines everything today's session is targeting." }];
  }

  // hardening #9: final repair pass
  selected = validateSessionComposition(selected, profile, lf);

  const ctx = { focusSkillIds, supportingSlot, maintenanceId, calibrationId };
  selected = selected.map((d) => (d.reason ? d : { ...d, reason: reasonForDrill(d, ctx) }));

  const startingRatings = Object.fromEntries(SKILLS.map((s) => [s.id, profile.skills[s.id].rating]));

  return { drills: selected, lf, weighting, focusSkillIds, startingRatings };
}

/* ---------------------------------------------------------------
   PROGRESS HISTORY
--------------------------------------------------------------- */
function appendRatingSnapshots(profile) {
  const history = { ...(profile.ratingHistory || {}) };
  SKILLS.forEach((s) => {
    const arr = history[s.id] ? [...history[s.id]] : [];
    const last = arr[arr.length - 1];
    const current = profile.skills[s.id].rating;
    if (!last || Math.round(last.rating) !== Math.round(current)) arr.push({ ts: Date.now(), rating: current });
    history[s.id] = arr.slice(-20);
  });
  return history;
}
function trendFor(history, skillId) {
  const arr = history?.[skillId];
  if (!arr || arr.length < 2) return "stable";
  const delta = arr[arr.length - 1].rating - arr[arr.length - 2].rating;
  if (delta > 1.5) return "up";
  if (delta < -1.5) return "down";
  return "stable";
}

/* ---------------------------------------------------------------
   SESSION SUMMARY — session-only tally vs long-term profile
   (hardening #12, #13)
--------------------------------------------------------------- */
function buildSummary(log, sessionState, finalProfile) {
  const skillDeltas = {};
  log.forEach((e) => {
    if (!e.skillId) return;
    skillDeltas[e.skillId] = skillDeltas[e.skillId] || { good: 0, bad: 0 };
    if (e.value >= 0.7) skillDeltas[e.skillId].good++;
    else skillDeltas[e.skillId].bad++;
  });
  const todayWentWell = Object.entries(skillDeltas).filter(([, v]) => v.good > v.bad).map(([id]) => SKILL_MAP[id].name);
  const todayLimited = Object.entries(skillDeltas).filter(([, v]) => v.bad > v.good).map(([id]) => SKILL_MAP[id].name);

  const chainNarratives = log.filter((e) => e.chainNarrative).map((e) => e.chainNarrative);
  const adaptations = log.filter((e) => e.type === "adaptation");

  const startingLf = sessionState.lf;
  const newLf = limitingFactor(finalProfile);
  const startingRatings = sessionState.startingRatings || {};

  let changeNote;
  const priorPrimaryId = startingLf.primary?.id;
  const priorImproved = priorPrimaryId && finalProfile.skills[priorPrimaryId].rating > (startingRatings[priorPrimaryId] ?? 0) + 3;
  const shifted = priorPrimaryId && newLf.primary && newLf.primary.id !== priorPrimaryId;

  if (priorPrimaryId && priorImproved && shifted) {
    changeNote = `${SKILL_MAP[priorPrimaryId].name} improved again today and is no longer your clearest weakness. Recent data now points to ${newLf.primary.name} as the main focus, so your next session shifts toward that while ${SKILL_MAP[priorPrimaryId].name.toLowerCase()} moves into maintenance.`;
  } else if (newLf.primary) {
    changeNote = `Your next session will continue focusing on ${newLf.primary.name}${newLf.secondary ? ` alongside some ${newLf.secondary.name}` : ""}.`;
  } else {
    changeNote = "Your next session will keep sampling broadly while confidence builds.";
  }

  return { todayWentWell, todayLimited, chainNarratives, adaptations, startingLf, newLf, changeNote, log };
}

/* ---------------------------------------------------------------
   ASSESSMENT — accumulates every item's update via a ref
--------------------------------------------------------------- */
function Assessment({ idx, setIdx, profile, onDone }) {
  const totalItems = ASSESSMENT_ITEMS.length + 1;
  const isClearance = idx === ASSESSMENT_ITEMS.length;
  const accumRef = useRef(profile);

  function advance(updatedProfile) {
    accumRef.current = updatedProfile;
    if (idx + 1 >= totalItems) onDone(accumRef.current);
    else setIdx(idx + 1);
  }

  return (
    <div>
      <div style={{ marginBottom: 14 }}>
        <Label>Step {idx + 1} of {totalItems}</Label>
        <div style={{ height: 6, background: T.panel2, borderRadius: 4, overflow: "hidden" }}>
          <div style={{ height: "100%", width: `${(idx / totalItems) * 100}%`, background: T.brass }} />
        </div>
      </div>
      {!isClearance ? (
        <DrillRunnerItem key={ASSESSMENT_ITEMS[idx].id} drill={ASSESSMENT_ITEMS[idx]} profile={accumRef.current} source="assessment" onComplete={(p) => advance(p)} />
      ) : (
        <ClearanceRunner key="assessment-clearance" clearance={ASSESSMENT_CLEARANCE} profile={accumRef.current} source="assessment" onComplete={(p) => advance(p)} />
      )}
    </div>
  );
}

  const radarData = SKILLS.map((s) => ({ skill: shortName(s.name), value: profile.skills[s.id].rating }));

/* ---------------------------------------------------------------
   SESSION RUNNER
--------------------------------------------------------------- */
function SessionRunner({ profile, sessionState, setSessionState, onFinish }) {
  const { drills, stepIdx } = sessionState;
  const current = drills[stepIdx];
  const logRef = useRef(sessionState.log || []);
  const profileRef = useRef(profile);

  function handleComplete(updatedProfile, entryOrEntries) {
    const newEntries = Array.isArray(entryOrEntries) ? entryOrEntries : [entryOrEntries];
    logRef.current = [...logRef.current, ...newEntries];
    profileRef.current = updatedProfile;
    if (stepIdx + 1 >= drills.length) {
      const summary = buildSummary(logRef.current, sessionState, updatedProfile);
      onFinish(summary, updatedProfile);
    } else {
      setSessionState((s) => ({ ...s, stepIdx: s.stepIdx + 1, log: logRef.current }));
    }
  }

  return (
    <div>
      <Label>Drill {stepIdx + 1} of {drills.length}</Label>
      <div style={{ height: 6, background: T.panel2, borderRadius: 4, overflow: "hidden", marginBottom: 14 }}>
        <div style={{ height: "100%", width: `${(stepIdx / drills.length) * 100}%`, background: T.chalk }} />
      </div>
      {current.type === "combined" ? (
        <ClearanceRunner key={`${current.id}-${stepIdx}`} clearance={current} profile={profileRef.current} source="training" onComplete={(p, entries) => handleComplete(p, entries)} />
      ) : (
        <DrillRunnerItem key={`${current.id}-${stepIdx}`} drill={current} profile={profileRef.current} source="training" onComplete={(p, entry) => handleComplete(p, entry)} />
      )}
    </div>
  );
}

/* ---------------------------------------------------------------
   DRILL RUNNER (pure execution / decision)
--------------------------------------------------------------- */
function DrillRunnerItem({ drill, profile, source = "training", onComplete }) {
  const [showErrorGrid, setShowErrorGrid] = useState(false);
  const [feedback, setFeedback] = useState(null); // { opt, updatedProfile }

  if (drill.type === "execution") {
    return (
      <Card>
        <div style={{ display: "flex", justifyContent: "center", marginBottom: 14 }}><TableDiagram balls={[{ color: T.chalk }, { color: "#D9C089" }]} /></div>
        <div style={{ fontSize: 17, fontWeight: 700, marginBottom: 4 }}>{drill.name}</div>
        <div style={{ fontSize: 13, color: T.inkDim, marginBottom: 6 }}>{drill.desc}</div>
        <div style={{ fontSize: 11, color: T.inkDim, marginBottom: 12 }}>Difficulty {drill.difficulty}/10</div>
        <WhyThisDrill reason={drill.reason} />
        {!showErrorGrid ? (
          <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 10 }}>
            <Btn kind="success" style={{ padding: "22px 0" }} onClick={() => {
              const updated = applySkillUpdate(profile, drill.skillId, 1, { drillId: drill.id, difficulty: drill.difficulty, source });
              onComplete(updated, { skillId: drill.skillId, value: 1 });
            }}><span style={{ display: "flex", alignItems: "center", justifyContent: "center", gap: 6 }}><Check size={20} /> Success</span></Btn>
            <Btn kind="fail" style={{ padding: "22px 0" }} onClick={() => setShowErrorGrid(true)}>
              <span style={{ display: "flex", alignItems: "center", justifyContent: "center", gap: 6 }}><X size={20} /> Failed</span></Btn>
          </div>
        ) : (
          <ErrorGrid onPick={(code) => {
            const updated = applySkillUpdate(profile, drill.skillId, 0, { drillId: drill.id, difficulty: drill.difficulty, reportedError: code, source });
            onComplete(updated, { skillId: drill.skillId, value: 0, reportedError: code });
          }} />
        )}
      </Card>
    );
  }

  // decision drill — scenario-specific options with rationale feedback
  if (!feedback) {
    return (
      <Card>
        <div style={{ display: "flex", justifyContent: "center", marginBottom: 14 }}><TableDiagram balls={[{ color: "#D9C089" }, { color: T.chalk }, { color: T.rust }]} /></div>
        <div style={{ fontSize: 17, fontWeight: 700, marginBottom: 4 }}>{drill.name}</div>
        <div style={{ fontSize: 13, color: T.inkDim, marginBottom: 12 }}>{drill.desc}</div>
        <WhyThisDrill reason={drill.reason} />
        <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
          {drill.options.map((option) => (
            <Btn key={option.key} style={{ textAlign: "left" }} onClick={() => {
              const val = decisionValue(option.tier);
              const updated = applySkillUpdate(profile, drill.skillId, val, { drillId: drill.id, difficulty: drill.difficulty, tier: option.tier, source });
              setFeedback({ option, updatedProfile: updated, value: val });
            }}>{option.label}</Btn>
          ))}
        </div>
      </Card>
    );
  }
  return (
    <Card>
      <Label>{feedback.option.tier === "optimal" ? "Strong choice" : feedback.option.tier === "acceptable" ? "Reasonable choice" : feedback.option.tier === "highrisk" ? "High risk" : "Not the best option"}</Label>
      <p style={{ margin: "0 0 16px", lineHeight: 1.5 }}>{feedback.option.rationale}</p>
      <Btn kind="primary" style={{ width: "100%" }} onClick={() => onComplete(feedback.updatedProfile, { skillId: drill.skillId, value: feedback.value, tier: feedback.option.tier })}>Continue</Btn>
    </Card>
  );
}

/* ---------------------------------------------------------------
   CLEARANCE RUNNER — unique-ID balls, real planned-vs-actual route,
   per-drill failure behaviour, decision evidence from real choices
--------------------------------------------------------------- */
function ClearanceRunner({ clearance, profile, source = "training", onComplete }) {
  const stateRef = useRef({
    profile, entries: [], remaining: clearance.balls.map((b) => b.id),
    actualRoute: [], plannedRoute: null, currentBallId: null,
  });
  const [, setTick] = useState(0);
  const rerender = () => setTick((t) => t + 1);
  const [showErrorGrid, setShowErrorGrid] = useState(false);
  const [showAdaptation, setShowAdaptation] = useState(false);
  const [planningPhase, setPlanningPhase] = useState(clearance.planEligible ? "choose" : "done");
  const [plannedOrder, setPlannedOrder] = useState(clearance.balls.map((b) => b.id));

  const st = stateRef.current;
  const ballById = Object.fromEntries(clearance.balls.map((b) => [b.id, b]));

  function shotEntries() {
    return st.entries.filter((e) => e.observedSkill);
  }

  function finishClearance() {
    const decisionResult = computeClearanceDecisionScore(st.entries);
    let finalProfile = st.profile;
    let finalEntries = st.entries;
    if (decisionResult) {
      finalProfile = applySkillUpdate(finalProfile, decisionResult.skillId, decisionResult.value, { drillId: clearance.id, difficulty: clearance.difficulty, tier: decisionResult.tier, clearance: true, source });
      finalEntries = [...finalEntries, { skillId: decisionResult.skillId, value: decisionResult.value, tier: decisionResult.tier, clearance: true }];
    }
    const chain = classifyErrorChain(shotEntries());
    if (chain && chain.rootSkill !== chain.immediateSkill) {
      finalProfile = bumpRootCauseTally(finalProfile, chain.rootSkill);
    }
    const narrative = buildErrorChainNarrative(chain);
    finalEntries = [...finalEntries, { meta: true, chainNarrative: narrative, plannedRoute: st.plannedRoute, actualRoute: st.actualRoute, errorChain: chain }];
    onComplete(finalProfile, finalEntries);
  }

  function afterShotContinue() {
    if (st.remaining.length === 0) finishClearance();
    else rerender();
  }

  function handleSelectBall(ballId) {
    st.currentBallId = ballId;
    rerender();
  }

  function handleResult(success) {
    const ball = ballById[st.currentBallId];
    if (success) {
      const updated = applySkillUpdate(st.profile, ball.execSkill, 1, { drillId: clearance.id, difficulty: clearance.difficulty, source, clearance: true, ballId: ball.id });
      st.profile = updated;
      st.entries = [...st.entries, { observedSkill: ball.execSkill, value: 1, ballId: ball.id }];
      st.actualRoute = [...st.actualRoute, ball.id];
      st.remaining = st.remaining.filter((id) => id !== ball.id);
      st.currentBallId = null;
      afterShotContinue();
    } else {
      setShowErrorGrid(true);
    }
  }

  function handleErrorPick(code) {
    setShowErrorGrid(false);
    const ball = ballById[st.currentBallId];
    // hardening #1: raw evidence ALWAYS updates the observed (task) skill — never redirected
    const updated = applySkillUpdate(st.profile, ball.execSkill, 0, {
      drillId: clearance.id, difficulty: clearance.difficulty, source, clearance: true, ballId: ball.id,
      reportedError: code, reportedMechanismSkill: ERROR_SKILL_MAP[code] || null,
    });
    st.profile = updated;
    st.entries = [...st.entries, { observedSkill: ball.execSkill, value: 0, ballId: ball.id, reportedError: code, reportedMechanismSkill: ERROR_SKILL_MAP[code] || null }];
    st.actualRoute = [...st.actualRoute, ball.id];

    const mode = resolveFailureMode(clearance, code);
    if (mode === "end_clearance") {
      st.currentBallId = null;
      finishClearance();
      return;
    }
    // ball stays available (reset_shot / continue_from_position / recovery_mode)
    st.currentBallId = null;
    if (code === "POSITION" && clearance.adaptationEligible) {
      setShowAdaptation(true);
    } else {
      afterShotContinue();
    }
  }

  function handleAdaptationChoice(choice) {
    setShowAdaptation(false);
    const preferred = clearance.preferredAdaptation;
    let tier = "acceptable";
    if (preferred && choice === preferred) tier = "optimal";
    else if (choice === "Continue original route" && preferred && preferred !== choice) tier = "poor";
    else if (choice === "Other") tier = "highrisk";
    const skillId = ADAPTATION_SKILL_MAP[choice] || "pattern";
    const value = decisionValue(tier);
    const updated = applySkillUpdate(st.profile, skillId, value, { drillId: clearance.id, difficulty: clearance.difficulty, tier, source: "adaptation" });
    st.profile = updated;
    st.entries = [...st.entries, { type: "adaptation", skillId, positionLost: true, recognised: true, choice, decisionTier: tier, value }];
    afterShotContinue();
  }

  if (planningPhase === "choose") {
    return (
      <Card>
        <div style={{ fontSize: 17, fontWeight: 700, marginBottom: 4 }}>{clearance.name}</div>
        <div style={{ fontSize: 13, color: T.inkDim, marginBottom: 16 }}>Would you like to plan this clearance first, or just play?</div>
        <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
          <Btn kind="primary" onClick={() => setPlanningPhase("planning")}>Plan the clearance</Btn>
          <Btn onClick={() => { st.plannedRoute = null; setPlanningPhase("done"); }}>Just play</Btn>
        </div>
      </Card>
    );
  }
  if (planningPhase === "planning") {
    return (
      <Card>
        <div style={{ fontSize: 17, fontWeight: 700, marginBottom: 4 }}>Plan Your Order</div>
        <div style={{ fontSize: 13, color: T.inkDim, marginBottom: 14 }}>Reorder if needed, then confirm.</div>
        <div style={{ display: "flex", flexDirection: "column", gap: 8, marginBottom: 14 }}>
          {plannedOrder.map((ballId, i) => (
            <div key={ballId} style={{ display: "flex", alignItems: "center", justifyContent: "space-between", background: T.panel2, borderRadius: 8, padding: "10px 12px" }}>
              <span>{i + 1}. {ballById[ballId].label}</span>
              <div style={{ display: "flex", gap: 6 }}>
                <Btn kind="ghost" style={{ padding: "4px 10px" }} onClick={() => { if (i === 0) return; const copy = [...plannedOrder]; [copy[i - 1], copy[i]] = [copy[i], copy[i - 1]]; setPlannedOrder(copy); }}>↑</Btn>
                <Btn kind="ghost" style={{ padding: "4px 10px" }} onClick={() => { if (i === plannedOrder.length - 1) return; const copy = [...plannedOrder]; [copy[i + 1], copy[i]] = [copy[i], copy[i + 1]]; setPlannedOrder(copy); }}>↓</Btn>
              </div>
            </div>
          ))}
        </div>
        <Btn kind="primary" style={{ width: "100%" }} onClick={() => {
          st.plannedRoute = plannedOrder;
          const tier = routeTier(plannedOrder, clearance);
          const value = decisionValue(tier);
          const updated = applySkillUpdate(st.profile, "pattern", value, { drillId: clearance.id, difficulty: clearance.difficulty, tier, source: "planDecision" });
          st.profile = updated;
          st.entries = [...st.entries, { type: "planDecision", skillId: "pattern", value, tier }];
          setPlanningPhase("done");
        }}>Confirm Plan</Btn>
      </Card>
    );
  }
  if (showAdaptation) {
    return (
      <Card>
        <div style={{ fontSize: 17, fontWeight: 700, marginBottom: 4 }}>Position lost</div>
        <div style={{ fontSize: 13, color: T.inkDim, marginBottom: 16 }}>What's your plan?</div>
        <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
          {["Re-plan clearance", "Continue original route", "Develop a problem ball", "Play safe", "Other"].map((c) => (
            <Btn key={c} onClick={() => handleAdaptationChoice(c)} style={{ textAlign: "left" }}>{c}</Btn>
          ))}
        </div>
      </Card>
    );
  }

  // ball-selection / shot phase
  if (!st.currentBallId) {
    if (st.remaining.length === 1) { st.currentBallId = st.remaining[0]; }
    else {
      return (
        <Card>
          <div style={{ display: "flex", justifyContent: "center", marginBottom: 10 }}>
            <TableDiagram balls={clearance.balls.map((b) => ({ color: st.remaining.includes(b.id) ? (b.group === "black" ? "#222" : b.group === "red" ? T.rust : "#D9C089") : T.panel2 }))} />
          </div>
          <div style={{ fontSize: 12, color: T.inkDim, marginBottom: 4 }}>{clearance.name}</div>
          <div style={{ fontSize: 17, fontWeight: 700, marginBottom: 12 }}>Which ball are you playing?</div>
          <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
            {st.remaining.map((id) => <Btn key={id} onClick={() => handleSelectBall(id)} style={{ textAlign: "left" }}>{ballById[id].label}</Btn>)}
          </div>
        </Card>
      );
    }
  }

  const ball = ballById[st.currentBallId];
  return (
    <Card>
      <div style={{ display: "flex", justifyContent: "center", marginBottom: 10 }}>
        <TableDiagram balls={clearance.balls.map((b) => ({ color: b.id === ball.id ? T.brass : st.remaining.includes(b.id) ? T.panel2 : T.green }))} />
      </div>
      <div style={{ fontSize: 12, color: T.inkDim, marginBottom: 4 }}>{clearance.name}</div>
      <div style={{ fontSize: 17, fontWeight: 700, marginBottom: 4 }}>Pot the {ball.label}</div>
      <div style={{ fontSize: 12, color: T.inkDim, marginBottom: 12 }}>Skill in focus: {SKILL_MAP[ball.execSkill].name}</div>
      <WhyThisDrill reason={clearance.reason} />
      {!showErrorGrid ? (
        <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 10 }}>
          <Btn kind="success" style={{ padding: "22px 0" }} onClick={() => handleResult(true)}><span style={{ display: "flex", alignItems: "center", justifyContent: "center", gap: 6 }}><Check size={20} /> Success</span></Btn>
          <Btn kind="fail" style={{ padding: "22px 0" }} onClick={() => handleResult(false)}><span style={{ display: "flex", alignItems: "center", justifyContent: "center", gap: 6 }}><X size={20} /> Failed</span></Btn>
        </div>
      ) : (
        <ErrorGrid onPick={handleErrorPick} />
      )}
    </Card>
  );
}
