3. ADD ONE DISPLAY HELPER
==================================================

Add a reusable helper in App.tsx or the existing UI helpers module:

function groupDisplayName(
  group?: "red" | "yellow" | "black" | "cue" | null
): string | null {
  if (!group) return null;

  switch (group) {
    case "red":
      return "REDS";
    case "yellow":
      return "YELLOWS";
    case "black":
      return "BLACK";
    case "cue":
      return "CUE";
    default:
      return null;
  }
}

Use this helper everywhere playerGroup is displayed.

Do not hard-code "YELLOWS" inside individual React drill branches.
